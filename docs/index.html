Quick Start & Cargo Features

Add rustgram and tokio to your project's Cargo.toml:

[dependencies]
rustgram = { git = "https://github.com/lethemery/RustGram" }
tokio = { version = "1", features = ["macros", "rt-multi-thread"] }

webhookYesrustgram::webhook module (axum webhook server, secret token guard, parser).
webappYesrustgram::webapp module (Telegram Mini App initData HMAC-SHA256 verification).

Minimal Echo Bot

use rustgram::prelude::*;

#[tokio::main]
async fn main() -> Result<()> {
    let bot = Bot::from_env()?;

    let router = Router::new()
        .route(on::command("start"), |ctx: Context| async move {
            ctx.reply(format!(
                "Hello! Running on rustgram (Bot API {})",
                rustgram::BOT_API_VERSION
            ))
            .await?;
            Ok(())
        })
        .route(on::text(), |ctx: Context| async move {
            if let Some(text) = ctx.message().and_then(|m| m.text.as_deref()) {
                ctx.reply(text).await?;
            }
            Ok(())
        });

    Dispatcher::new(bot, router).run_polling().await;
    Ok(())
}

Client, BotBuilder, Files & Error Handling

Bot is cheaply cloneable (backed by Arc) and exposes every Bot API request as a builder implementing IntoFuture<Output = Result<T>>. Calling .send().await or simply .await dispatches the JSON or multipart HTTP request. 

use std::time::Duration;
use rustgram::prelude::*;
use rustgram::types::InputFile;

async fn setup_and_send(token: &str, chat_id: i64) -> Result<()> {
    let bot = Bot::builder(token)
        .base_url("https://api.telegram.org") // or self-hosted Local Bot API server
        .test_environment(false)
        .timeout(Duration::from_secs(60))
        .retry_rate_limits(3, Duration::from_secs(15))
        .build()?;

    let me = bot.get_me().await?;
    println!("Connected as @{}", me.username.unwrap_or_default());

    // Upload an in-memory file via multipart/form-data
    let doc = InputFile::memory("report.txt", b"Bot API 10.4 ready".to_vec());
    bot.send_document(chat_id, doc)
        .caption("Generated status report")
        .await?;

    Ok(())
}

Router, Filters, Middleware & Dependency Injection

Router evaluates routes top-to-bottom. Combine built-in filters in rustgram::router::on with .and(), .or(), !filter, or .negated(), write custom predicates using filter_fn / async_filter, group sub-routers via .nest(), attach cross-cutting middleware via .layer(), and inject shared dependencies via Dispatcher::manage(). 

use std::sync::Arc;
use rustgram::prelude::*;
use tokio::sync::Mutex;

#[derive(Default)]
struct Metrics {
    handled: Mutex<u64>,
}

fn build_router(metrics: Arc<Metrics>) -> Dispatcher {
    let bot = Bot::from_env().expect("TELEGRAM_BOT_TOKEN");

    let admin_router = Router::new()
        .route(on::command("stats"), |ctx: Context| async move {
            let stats = ctx.state::<Arc<Metrics>>().unwrap();
            let total = *stats.handled.lock().await;
            ctx.reply(format!("Updates processed: {total}")).await?;
            Ok(())
        });

    let root = Router::new()
        .layer(|ctx: Context, next: Next| async move {
            if let Some(m) = ctx.state::<Arc<Metrics>>() {
                *m.handled.lock().await += 1;
            }
            next.run(ctx).await
        })
        .route(on::callback_prefix("buy:"), |ctx: Context| async move {
            ctx.answer_callback()?.text("Order queued").await?;
            Ok(())
        })
        .nest(on::private_chat(), admin_router)
        .fallback(|ctx: Context| async move {
            let _ = ctx.reply("Try /stats in private chat.").await;
            Ok(())
        });

    Dispatcher::new(bot, root).manage(metrics)
}

Роутер, фильтры, Middleware и DI

Router проверяет маршруты сверху вниз. Комбинируйте фильтры из rustgram::router::on через .and(), .or(), !filter или .negated(), пишите свои условия на filter_fn / async_filter, вкладывайте под-роутеры через .nest(), добавляйте middleware через .layer() и передавайте общие сервисы через Dispatcher::manage(). 

use std::sync::Arc;
use rustgram::prelude::*;
use tokio::sync::Mutex;

#[derive(Default)]
struct Metrics {
    handled: Mutex<u64>,
}

fn build_router(metrics: Arc<Metrics>) -> Dispatcher {
    let bot = Bot::from_env().expect("TELEGRAM_BOT_TOKEN");

    let admin_router = Router::new()
        .route(on::command("stats"), |ctx: Context| async move {
            let stats = ctx.state::<Arc<Metrics>>().unwrap();
            let total = *stats.handled.lock().await;
            ctx.reply(format!("Обработано апдейтов: {total}")).await?;
            Ok(())
        });

    let root = Router::new()
        .layer(|ctx: Context, next: Next| async move {
            if let Some(m) = ctx.state::<Arc<Metrics>>() {
                *m.handled.lock().await += 1;
            }
            next.run(ctx).await
        })
        .route(on::callback_prefix("buy:"), |ctx: Context| async move {
            ctx.answer_callback()?.text("Заказ принят").await?;
            Ok(())
        })
        .nest(on::private_chat(), admin_router)
        .fallback(|ctx: Context| async move {
            let _ = ctx.reply("Напишите /stats в личном чате.").await;
            Ok(())
        });

    Dispatcher::new(bot, root).manage(metrics)
}

Finite State Machine (FSM)

Multi-step conversational flows use StateStorage<S> (keyed by StateKey::new(chat_id, user_id)) and the per-request helper ctx.fsm::<S>()?. Built-in MemoryStorage<S> stores both the enum state and a JSON scratchpad map (.insert() / .get_value()). Implement StateStorage<S> for Redis, PostgreSQL, or SQLite backends when scaling horizontally. 

use std::sync::Arc;
use rustgram::prelude::*;

#[derive(Debug, Clone, PartialEq, Eq)]
enum OrderStep {
    AskTitle,
    AskAmount,
}

fn fsm_router(bot: Bot) -> Dispatcher {
    let storage: Arc<dyn StateStorage<OrderStep>> = Arc::new(MemoryStorage::new());

    let router = Router::new()
        .route(on::command("order"), |ctx: Context| async move {
            ctx.fsm::<OrderStep>()?.set(OrderStep::AskTitle).await?;
            ctx.reply("Step 1/2: Send item title:").await?;
            Ok(())
        })
        .route(on::state(OrderStep::AskTitle).and(on::text()), |ctx: Context| async move {
            let title = ctx.message().and_then(|m| m.text.as_deref()).unwrap_or("");
            let fsm = ctx.fsm::<OrderStep>()?;
            fsm.insert("title", title).await?;
            fsm.set(OrderStep::AskAmount).await?;
            ctx.reply("Step 2/2: Enter quantity:").await?;
            Ok(())
        })
        .route(on::state(OrderStep::AskAmount).and(on::text()), |ctx: Context| async move {
            let amount = ctx.message().and_then(|m| m.text.as_deref()).unwrap_or("1");
            let fsm = ctx.fsm::<OrderStep>()?;
            let title: String = fsm.get_value("title").await?.unwrap_or_default();
            fsm.clear().await?;
            ctx.reply(format!("Order created: {title} x {amount}")).await?;
            Ok(())
        });

    Dispatcher::new(bot, router).manage_fsm(storage)
}

Конечный автомат (FSM)

Пошаговые диалоги строятся на трейте StateStorage<S> (ключ StateKey::new(chat_id, user_id)) и хелпере контекста ctx.fsm::<S>()?. Встроенный MemoryStorage<S> хранит текущее состояние и JSON-словарь данных (.insert() / .get_value()). Для продакшена с несколькими инстансами достаточно реализовать StateStorage<S> поверх Redis или SQL. 

use std::sync::Arc;
use rustgram::prelude::*;

#[derive(Debug, Clone, PartialEq, Eq)]
enum OrderStep {
    AskTitle,
    AskAmount,
}

fn fsm_router(bot: Bot) -> Dispatcher {
    let storage: Arc<dyn StateStorage<OrderStep>> = Arc::new(MemoryStorage::new());

    let router = Router::new()
        .route(on::command("order"), |ctx: Context| async move {
            ctx.fsm::<OrderStep>()?.set(OrderStep::AskTitle).await?;
            ctx.reply("Шаг 1/2: Введите название товара:").await?;
            Ok(())
        })
        .route(on::state(OrderStep::AskTitle).and(on::text()), |ctx: Context| async move {
            let title = ctx.message().and_then(|m| m.text.as_deref()).unwrap_or("");
            let fsm = ctx.fsm::<OrderStep>()?;
            fsm.insert("title", title).await?;
            fsm.set(OrderStep::AskAmount).await?;
            ctx.reply("Шаг 2/2: Укажите количество:").await?;
            Ok(())
        })
        .route(on::state(OrderStep::AskAmount).and(on::text()), |ctx: Context| async move {
            let amount = ctx.message().and_then(|m| m.text.as_deref()).unwrap_or("1");
            let fsm = ctx.fsm::<OrderStep>()?;
            let title: String = fsm.get_value("title").await?.unwrap_or_default();
            fsm.clear().await?;
            ctx.reply(format!("Заказ оформлен: {title} x {amount}")).await?;
            Ok(())
        });

    Dispatcher::new(bot, router).manage_fsm(storage)
}

HTML Widgets & TON Wallet Integration (Bot API 10.4)

Bot API 10.4 introduces interactive in-chat HTML widgets and native TON wallet operations. rustgram validates HTML/data byte bounds and NUL characters locally before hitting the network, and provides typed variants for every WidgetQueryType callback.

Method / ConstantDetails
send_widget(chat_id, html)Sends an HTML widget with .size(w, h), .shared(bool), .public_data(), .chat_data(), .cache_time(), .ephemeral().
send_shared_widget(chat_id, widget_id)Reuses an existing shared widget identifier created with shared(true).
answer_widget_query(query_id)Replies to user widget interaction with optional updated .data(...).
edit_message_widget_data / edit_inline_widget_data / edit_shared_widget_dataUpdates public_data or chat_data of live message, inline, or shared widgets.
WIDGET_HTML_MAX_BYTES / WIDGET_DATA_MAX_BYTES32,768 bytes HTML limit, 1,024 bytes data payload limit, clamp_widget_cache_time clamps [5..=3600].
 
use rustgram::prelude::*;

fn widget_routes() -> Router {
    Router::new()
        .route(on::command("pay_widget"), |ctx: Context| async move {
            let html = r#"<button onclick="TelegramWidget.request('pay')">Pay 1 TON</button>"#;
            ctx.send_widget(html)?
                .size(320, 160)
                .shared(true)
                .cache_time(60)
                .await?;
            Ok(())
        })
        .route(on::any_widget_query(), |ctx: Context| async move {
            if let Some(wq) = ctx.widget_query() {
                match &wq.query_type {
                    WidgetQueryType::SentTransaction { wallet, msg_hash } => {
                        let _addr = &wallet.address;
                        ctx.answer_widget()?
                            .data(format!(r#"{{"paid":true,"tx":"{msg_hash}"}}"#))
                            .await?;
                    }
                    WidgetQueryType::SignData { wallet, sign_data } => {
                        let _ = (wallet, sign_data);
                        ctx.answer_widget()?.data(r#"{"signed":true}"#).await?;
                    }
                    WidgetQueryType::Data { with_wallet, wallet } => {
                        let _ = (with_wallet, wallet);
                        ctx.answer_widget()?.data(r#"{"status":"ok"}"#).await?;
                    }
                    WidgetQueryType::Unknown => {}
                }
            }
            Ok(())
        })
}

Rich Messages & Stoppable Streaming Drafts (Bot API 10.1–10.3)

Build structured multi-block layouts with InputRichMessage (HTML, Markdown, or explicit Vec<RichBlock>) and stream real-time LLM responses using send_message_draft or send_rich_message_draft with user-cancellable Stop controls. 

use rustgram::prelude::*;
use rustgram::types::{InputRichMessage, RichBlock, RichBlockTableCell, RichMessageButton};

async fn stream_and_finalize(bot: &Bot, chat_id: i64, draft_id: i64) -> Result<()> {
    // 1. Stream stoppable thinking/partial draft
    bot.send_message_draft(chat_id, draft_id, "Analyzing Bot API 10.4...")
        .can_stop(true)
        .keep_on_stop(true)
        .await?;

    // 2. Finalize with rich semantic blocks
    let rich = InputRichMessage::blocks(vec![
        RichBlock::SectionHeading {
            text: "Analysis Complete".into(),
            level: Some(2),
            entities: None,
        },
        RichBlock::Table {
            caption: Some("Bot API 10.4 Summary".into()),
            rows: vec![
                vec![
                    RichBlockTableCell::header("Module"),
                    RichBlockTableCell::header("Version"),
                ],
                vec![
                    RichBlockTableCell::text("Widgets"),
                    RichBlockTableCell::text("10.4"),
                ],
            ],
            is_compact: Some(true),
        },
        RichBlock::Buttons {
            buttons: vec![vec![
                RichMessageButton::callback("Confirm", "ok"),

Ephemeral Messages, Guest Mode & Managed Bots (9.4–10.4)

Modern Bot API features across versions 9.4 through 10.4 are modeled as first-class types and request methods:

  • Ephemeral Messages (10.2–10.4): Use EphemeralMessageParameters::for_callback(cq_id) without receiver_user_id (added in 10.4) and .replace_callback_message(true), mark commands with BotCommand::new(cmd, desc).ephemeral(true), or call edit_ephemeral_message_* / delete_ephemeral_message.
  • Guest Mode (10.2): Route on::guest_message() and respond with ctx.answer_guest() or bot.answer_guest_query(...).
  • Managed Bots (9.6–10.4): Request bot creation via KeyboardButton::request_managed_bot(...) with custom emoji icons (.icon_custom_emoji(...)), handle on::managed_bot(), and manage tokens/access via get_managed_bot_token, replace_managed_bot_token, get_managed_bot_access_settings, and set_managed_bot_access_settings.
  • Communities, Subscriptions & Entities: Handle on::subscription() (BotSubscriptionUpdated), set_chat_member_tag, answer_chat_join_request_query, get_user_personal_chat_messages, create_forum_topic, and date_time entities.
use rustgram::prelude::*;
use rustgram::types::{BotCommand, EphemeralMessageParameters, KeyboardButton, KeyboardButtonRequestManagedBot};

async fn send_ephemeral_and_commands(bot: &Bot, callback_query_id: &str, chat_id: i64) -> Result<()> {
    // Ephemeral command in chat menu
    bot.set_my_commands(vec![
        BotCommand::new("secret", "Only visible to caller").ephemeral(true),
    ])
    .await?;

    // Callback-targeted ephemeral message (10.4: no receiver_user_id required)
    let eph = EphemeralMessageParameters::for_callback(callback_query_id)
        .replace_callback_message(true);
    bot.send_message(chat_id, "Private ephemeral confirmation")
        .ephemeral(eph)
        .await?;

    // Managed bot button with custom emoji icon
    let _btn = KeyboardButton::request_managed_bot(
        "Create Helper Bot",
        KeyboardButtonRequestManagedBot::new(1).suggested_name("Helper"),
    )
    .icon_custom_emoji("5368324170671202286");

    Ok(())
}

Эфемерные сообщения, Guest Mode и Managed Bots (9.4–10.4)

Все ключевые возможности Bot API с версии 9.4 по 10.4 реализованы в виде строгих типов и методов:

  • Эфемерные сообщения (10.2–10.4): EphemeralMessageParameters::for_callback(cq_id) работает без receiver_user_id (нововведение 10.4) и поддерживает .replace_callback_message(true); команды помечаются BotCommand::new(cmd, desc).ephemeral(true), доступны edit_ephemeral_message_* и delete_ephemeral_message.

  • Управляемые боты (9.6–10.4): Кнопки KeyboardButton::request_managed_bot(...) с иконкой .icon_custom_emoji(...), обновления on::managed_bot() и методы get_managed_bot_token, replace_managed_bot_token, get_managed_bot_access_settings, set_managed_bot_access_settings.
  • Сообщества, подписки и сущности: Поддержка on::subscription() (BotSubscriptionUpdated), set_chat_member_tag, answer_chat_join_request_query, get_user_personal_chat_messages, create_forum_topic и date_time.

Polling, Axum Webhooks, Mini Apps & Formatting

Choose between resilient long polling (with exponential backoff and graceful shutdown via CancellationToken) or an axum webhook server, plus helpers for Mini App signature verification and safe markup.

use std::net::SocketAddr;
use rustgram::formatting::{bold_html, date_time_html, escape_html};
use rustgram::prelude::*;
use rustgram::webapp::verify_init_data;
use rustgram::webhook::{WebhookConfig, serve_webhook};

async fn run_webhook_and_helpers(bot: Bot, router: Router) -> Result<()> {
    // Safe HTML formatting including Telegram date_time tags
    let formatted = format!(
        "{} — {}",
        bold_html("Status <ok>"),
        date_time_html(1794240000, Some("wDT"), "Release Day")
    );
    let _escaped = escape_html(&formatted);

    // Verify Telegram Mini App initData payload (enabled via `webapp` feature)
    // let verified = verify_init_data(init_data_raw, &token, Some(300), None)?;

    // Run Axum webhook endpoint (enabled via `webhook` feature)
    let addr: SocketAddr = "0.0.0.0:8080".parse().unwrap();
    let cfg = WebhookConfig {
        path: "/tg/webhook".into(),
        secret_token: Some("my-webhook-secret".into()),
        concurrent_handling: true,
    };
    let dp = Dispatcher::new(bot, router);
    serve_webhook(addr, cfg, move |update| {
        let dp = dp.clone();
        async move {
            let _ = dp.dispatch(update).await;
        }
    })
    .await?;

    let _ = verify_init_data;
    Ok(())
}
